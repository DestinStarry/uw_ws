cmake_minimum_required(VERSION 3.16)
project(uw_robot_control)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
find_package(rclcpp REQUIRED)
find_package(uw_robot_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)

# closed_loop: private algorithm/protocol library and independent node.
add_library(closed_loop_logic STATIC src/closed_loop/controller.cpp)
target_compile_features(closed_loop_logic PUBLIC cxx_std_17)
target_compile_options(closed_loop_logic PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(closed_loop_logic PUBLIC include)
target_link_libraries(closed_loop_logic Eigen3::Eigen)
ament_target_dependencies(closed_loop_logic uw_robot_interfaces)
add_executable(control_node src/closed_loop/control_node.cpp)
target_compile_features(control_node PRIVATE cxx_std_17)
target_compile_options(control_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(control_node rclcpp uw_robot_interfaces sensor_msgs)
target_link_libraries(control_node closed_loop_logic Threads::Threads)
install(TARGETS control_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(control_test test/closed_loop/control_test.cpp)
  target_link_libraries(control_test closed_loop_logic)
  add_test(NAME control_test COMMAND control_test)
  add_executable(mode_switch_test test/closed_loop/mode_switch_test.cpp)
  target_link_libraries(mode_switch_test closed_loop_logic)
  add_test(NAME mode_switch_test COMMAND mode_switch_test)
  add_executable(control_cli test/closed_loop/control_cli.cpp)
  target_link_libraries(control_cli closed_loop_logic)
endif()

# allocation: private algorithm/protocol library and independent node.
add_library(allocation_logic STATIC src/allocation/allocator.cpp src/allocation/allocation_controller.cpp)
target_compile_features(allocation_logic PUBLIC cxx_std_17)
target_compile_options(allocation_logic PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(allocation_logic PUBLIC include)
target_link_libraries(allocation_logic Eigen3::Eigen)
ament_target_dependencies(allocation_logic uw_robot_interfaces)
add_executable(allocation_node src/allocation/allocation_node.cpp)
target_compile_features(allocation_node PRIVATE cxx_std_17)
target_compile_options(allocation_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(allocation_node rclcpp uw_robot_interfaces sensor_msgs)
target_link_libraries(allocation_node allocation_logic Threads::Threads)
install(TARGETS allocation_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(allocation_test test/allocation/allocation_test.cpp)
  target_link_libraries(allocation_test allocation_logic)
  add_test(NAME allocation_test COMMAND allocation_test)
  add_executable(parity_cli test/allocation/parity_cli.cpp)
  target_link_libraries(parity_cli allocation_logic)
endif()

ament_package()
